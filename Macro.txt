#pragma once
// The Nfbot ".ng" macro format + playback cursor. Pure C++, no Geode.
//
// A .ng file is plain JSON that carries everything about a run:
//   {
//     "format": "nfbot", "version": 1,
//     "meta":     { author, level {id, name}, game, mod, created },
//     "settings": { tps },
//     "inputs":   [ {"f": frame, "b": button 1..3, "d": down?, "p": player2?}, ... ]
//     ...any extra top-level keys you add are preserved on load/save
//   }
#include "Json.hpp"

#include <algorithm>
#include <climits>
#include <ctime>
#include <string>
#include <string_view>
#include <vector>

namespace nf {

struct Input {
    int frame = 0;
    int button = 1;  // 1 = jump, 2 = left, 3 = right
    bool down = false;
    bool player2 = false;
    bool operator==(const Input&) const = default;
};

inline std::string isoNow() {
    std::time_t t = std::time(nullptr);
    std::tm tm{};
#ifdef _WIN32
    gmtime_s(&tm, &t);
#else
    gmtime_r(&t, &tm);
#endif
    char buf[32];
    std::strftime(buf, sizeof buf, "%Y-%m-%dT%H:%M:%SZ", &tm);
    return buf;
}

struct Macro {
    static constexpr const char* kFormat = "nfbot";
    static constexpr int kVersion = 1;

    std::string author;
    std::string levelName;
    std::string gameVersion;
    std::string modVersion;
    std::string createdAt;
    int levelId = 0;
    double tps = 240.0;
    std::vector<Input> inputs;  // always sorted by frame (stable)
    Json extra = Json::object();  // unknown top-level keys, kept verbatim

    // Adds an input keeping the list sorted. Appending (the normal case) is O(1).
    void addInput(int frame, int button, bool down, bool player2) {
        Input in{frame, button, down, player2};
        if (inputs.empty() || inputs.back().frame <= frame) {
            inputs.push_back(in);
            return;
        }
        auto it = std::upper_bound(inputs.begin(), inputs.end(), frame,
                                   [](int f, const Input& x) { return f < x.frame; });
        inputs.insert(it, in);
    }

    // Removes every input with frame >= `frame` (used when a checkpoint is loaded).
    void truncateFrom(int frame) {
        auto it = std::lower_bound(inputs.begin(), inputs.end(), frame,
                                   [](const Input& x, int f) { return x.frame < f; });
        inputs.erase(it, inputs.end());
    }

    Json toJson() const {
        Json level = Json::object();
        level.set("id", levelId);
        level.set("name", levelName);

        Json meta = Json::object();
        meta.set("author", author);
        meta.set("level", std::move(level));
        meta.set("game", gameVersion);
        meta.set("mod", modVersion);
        meta.set("created", createdAt);

        Json settings = Json::object();
        settings.set("tps", tps);

        Json list = Json::array();
        for (const Input& in : inputs) {
            Json o = Json::object();
            o.set("f", in.frame);
            o.set("b", in.button);
            o.set("d", in.down);
            o.set("p", in.player2);
            list.push(std::move(o));
        }

        Json root = Json::object();
        root.set("format", kFormat);
        root.set("version", kVersion);
        root.set("meta", std::move(meta));
        root.set("settings", std::move(settings));
        root.set("inputs", std::move(list));
        for (const auto& [k, v] : extra.members()) {
            if (!isKnownKey(k)) root.set(k, v);
        }
        return root;
    }

    std::string serialize() const { return toJson().dump(2) + "\n"; }

    // Returns false and fills `error` on any problem; `out` is left untouched then.
    static bool deserialize(std::string_view text, Macro& out, std::string& error) {
        auto parsed = Json::parse(text);
        if (!parsed.ok) {
            error = "invalid JSON: " + parsed.error;
            return false;
        }
        const Json& root = parsed.value;
        if (!root.isObject()) {
            error = "root must be an object";
            return false;
        }
        const Json* fmt = root.find("format");
        if (!fmt || !fmt->isString() || fmt->asString() != kFormat) {
            error = "not an Nfbot macro (missing \"format\": \"nfbot\")";
            return false;
        }
        const Json* ver = root.find("version");
        if (!ver || !ver->isInteger() || ver->asInt() < 1) {
            error = "missing or invalid \"version\"";
            return false;
        }
        if (ver->asInt() > kVersion) {
            error = "macro uses format v" + std::to_string(ver->asInt()) +
                    ", this Nfbot only understands up to v" + std::to_string(kVersion);
            return false;
        }

        Macro m;
        if (const Json* meta = root.find("meta"); meta && meta->isObject()) {
            m.author = str(meta, "author");
            m.gameVersion = str(meta, "game");
            m.modVersion = str(meta, "mod");
            m.createdAt = str(meta, "created");
            if (const Json* level = meta->find("level"); level && level->isObject()) {
                m.levelName = str(level, "name");
                if (const Json* id = level->find("id"); id && id->isInteger()) m.levelId = id->asInt();
            }
        }
        if (const Json* settings = root.find("settings"); settings && settings->isObject()) {
            if (const Json* tps = settings->find("tps")) {
                if (!tps->isNumber() || !(tps->asNumber() > 0.0) || tps->asNumber() > 100000.0) {
                    error = "invalid \"tps\"";
                    return false;
                }
                m.tps = tps->asNumber();
            }
        }
        if (const Json* list = root.find("inputs")) {
            if (!list->isArray()) {
                error = "\"inputs\" must be an array";
                return false;
            }
            size_t idx = 0;
            for (const Json& j : list->items()) {
                std::string where = "input #" + std::to_string(idx++) + ": ";
                if (!j.isObject()) {
                    error = where + "not an object";
                    return false;
                }
                const Json* f = j.find("f");
                const Json* b = j.find("b");
                const Json* d = j.find("d");
                const Json* p = j.find("p");
                if (!f || !f->isInteger() || f->asInt() < 0) {
                    error = where + "\"f\" must be a non-negative integer";
                    return false;
                }
                if (!b || !b->isInteger() || b->asInt() < 1 || b->asInt() > 3) {
                    error = where + "\"b\" must be 1, 2 or 3";
                    return false;
                }
                if (!d || !d->isBool()) {
                    error = where + "\"d\" must be true or false";
                    return false;
                }
                if (p && !p->isBool()) {
                    error = where + "\"p\" must be true or false";
                    return false;
                }
                m.inputs.push_back(Input{f->asInt(), b->asInt(), d->asBool(), p ? p->asBool() : false});
            }
            std::stable_sort(m.inputs.begin(), m.inputs.end(),
                             [](const Input& a, const Input& b2) { return a.frame < b2.frame; });
        }
        for (const auto& [k, v] : root.members()) {
            if (!isKnownKey(k)) m.extra.set(k, v);
        }
        out = std::move(m);
        return true;
    }

private:
    static bool isKnownKey(const std::string& k) {
        return k == "format" || k == "version" || k == "meta" || k == "settings" || k == "inputs";
    }
    static std::string str(const Json* obj, const char* key) {
        const Json* v = obj->find(key);
        return v && v->isString() ? v->asString() : std::string();
    }
};

// Walks a macro during playback. Holds only an index, so it stays valid as long
// as the Macro object itself lives (the Geode glue keeps one static Macro).
class Cursor {
public:
    explicit Cursor(const Macro& macro) : m_macro(&macro) {}

    // Next dispatched input = first one whose (frame + offset) >= `frame`.
    void seek(int frame, int offset) {
        const auto& in = m_macro->inputs;
        auto it = std::partition_point(in.begin(), in.end(),
                                       [&](const Input& x) { return x.frame + offset < frame; });
        m_index = static_cast<size_t>(it - in.begin());
    }

    // Calls fn(input) for every pending input due at or before `frame`.
    template <class Fn>
    int dispatchDue(int frame, int offset, Fn&& fn) {
        const auto& in = m_macro->inputs;
        int count = 0;
        while (m_index < in.size() && in[m_index].frame + offset <= frame) {
            fn(in[m_index]);
            ++m_index;
            ++count;
        }
        return count;
    }

    bool finished() const { return m_index >= m_macro->inputs.size(); }

private:
    const Macro* m_macro;
    size_t m_index = 0;
};

}  // namespace nf
