#pragma once
// Tiny, dependency-free JSON value/parser/writer used by the .ng macro format.
// Kept independent of Geode so it can be unit-tested on any machine.
//  - objects keep insertion order (stable, diff-friendly files)
//  - parser is strict about structure, depth-limited, and never throws
#include <cmath>
#include <cstddef>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <string>
#include <string_view>
#include <type_traits>
#include <utility>
#include <vector>

namespace nf {

class Json {
public:
    enum class Type { Null, Bool, Number, String, Array, Object };
    using Array = std::vector<Json>;
    using Object = std::vector<std::pair<std::string, Json>>;

    struct ParseResult;

    Json() = default;
    Json(std::nullptr_t) {}
    Json(bool v) : m_type(Type::Bool), m_bool(v) {}
    template <class T, std::enable_if_t<std::is_arithmetic_v<T> && !std::is_same_v<T, bool>, int> = 0>
    Json(T v) : m_type(Type::Number), m_num(static_cast<double>(v)) {}
    Json(const char* v) : m_type(Type::String), m_str(v) {}
    Json(std::string v) : m_type(Type::String), m_str(std::move(v)) {}

    static Json array() {
        Json j;
        j.m_type = Type::Array;
        return j;
    }
    static Json object() {
        Json j;
        j.m_type = Type::Object;
        return j;
    }

    Type type() const { return m_type; }
    bool isNull() const { return m_type == Type::Null; }
    bool isBool() const { return m_type == Type::Bool; }
    bool isNumber() const { return m_type == Type::Number; }
    bool isString() const { return m_type == Type::String; }
    bool isArray() const { return m_type == Type::Array; }
    bool isObject() const { return m_type == Type::Object; }

    // True only for numbers with no fractional part that fit an int.
    bool isInteger() const {
        return m_type == Type::Number && std::isfinite(m_num) && std::floor(m_num) == m_num &&
               m_num >= -2147483648.0 && m_num <= 2147483647.0;
    }

    bool asBool(bool def = false) const { return m_type == Type::Bool ? m_bool : def; }
    double asNumber(double def = 0.0) const { return m_type == Type::Number ? m_num : def; }
    int asInt(int def = 0) const { return isInteger() ? static_cast<int>(m_num) : def; }
    const std::string& asString() const {
        static const std::string empty;
        return m_type == Type::String ? m_str : empty;
    }

    const Array& items() const { return m_arr; }
    const Object& members() const { return m_obj; }

    const Json* find(std::string_view key) const {
        if (m_type != Type::Object) return nullptr;
        for (const auto& [k, v] : m_obj) {
            if (k == key) return &v;
        }
        return nullptr;
    }

    // Sets (or replaces) a key. Turns a null value into an object first.
    Json& set(std::string key, Json value) {
        if (m_type == Type::Null) m_type = Type::Object;
        for (auto& [k, v] : m_obj) {
            if (k == key) {
                v = std::move(value);
                return v;
            }
        }
        m_obj.emplace_back(std::move(key), std::move(value));
        return m_obj.back().second;
    }

    void push(Json value) {
        if (m_type == Type::Null) m_type = Type::Array;
        m_arr.push_back(std::move(value));
    }

    // indent <= 0 -> compact. Containers holding only scalars stay on one line,
    // so a big "inputs" list is one input per line and easy to read/diff.
    std::string dump(int indent = 2) const {
        std::string out;
        write(out, indent, 0);
        return out;
    }

    static ParseResult parse(std::string_view text);

private:
    bool isScalar() const { return m_type != Type::Array && m_type != Type::Object; }

    bool isFlat() const {
        if (m_type == Type::Array) {
            for (const auto& c : m_arr) {
                if (!c.isScalar()) return false;
            }
            return true;
        }
        if (m_type == Type::Object) {
            for (const auto& [k, c] : m_obj) {
                if (!c.isScalar()) return false;
            }
            return true;
        }
        return true;
    }

    static void writeString(std::string& out, const std::string& s) {
        out.push_back('"');
        for (unsigned char c : s) {
            switch (c) {
                case '"': out += "\\\""; break;
                case '\\': out += "\\\\"; break;
                case '\n': out += "\\n"; break;
                case '\r': out += "\\r"; break;
                case '\t': out += "\\t"; break;
                case '\b': out += "\\b"; break;
                case '\f': out += "\\f"; break;
                default:
                    if (c < 0x20) {
                        char buf[8];
                        std::snprintf(buf, sizeof buf, "\\u%04x", c);
                        out += buf;
                    } else {
                        out.push_back(static_cast<char>(c));
                    }
            }
        }
        out.push_back('"');
    }

    static void writeNumber(std::string& out, double v) {
        if (!std::isfinite(v)) {
            out += "null";
            return;
        }
        char buf[40];
        if (std::floor(v) == v && std::fabs(v) < 1e15) {
            std::snprintf(buf, sizeof buf, "%lld", static_cast<long long>(v));
        } else {
            std::snprintf(buf, sizeof buf, "%.17g", v);
        }
        out += buf;
    }

    void write(std::string& out, int indent, int level) const {
        const bool pretty = indent > 0;
        auto pad = [&](int lv) {
            if (pretty) out.append(static_cast<size_t>(lv * indent), ' ');
        };
        switch (m_type) {
            case Type::Null: out += "null"; return;
            case Type::Bool: out += m_bool ? "true" : "false"; return;
            case Type::Number: writeNumber(out, m_num); return;
            case Type::String: writeString(out, m_str); return;
            case Type::Array: {
                if (m_arr.empty()) {
                    out += "[]";
                    return;
                }
                const bool inlineIt = !pretty || isFlat();
                out.push_back('[');
                for (size_t i = 0; i < m_arr.size(); ++i) {
                    if (i) out += inlineIt ? (pretty ? ", " : ",") : ",";
                    if (!inlineIt) {
                        out.push_back('\n');
                        pad(level + 1);
                    }
                    m_arr[i].write(out, indent, level + 1);
                }
                if (!inlineIt) {
                    out.push_back('\n');
                    pad(level);
                }
                out.push_back(']');
                return;
            }
            case Type::Object: {
                if (m_obj.empty()) {
                    out += "{}";
                    return;
                }
                const bool inlineIt = !pretty || isFlat();
                out.push_back('{');
                for (size_t i = 0; i < m_obj.size(); ++i) {
                    if (i) out += inlineIt ? (pretty ? ", " : ",") : ",";
                    if (!inlineIt) {
                        out.push_back('\n');
                        pad(level + 1);
                    }
                    writeString(out, m_obj[i].first);
                    out += pretty ? ": " : ":";
                    m_obj[i].second.write(out, indent, level + 1);
                }
                if (!inlineIt) {
                    out.push_back('\n');
                    pad(level);
                }
                out.push_back('}');
                return;
            }
        }
    }

    Type m_type = Type::Null;
    bool m_bool = false;
    double m_num = 0.0;
    std::string m_str;
    Array m_arr;
    Object m_obj;
};

struct Json::ParseResult {
    bool ok = false;
    Json value;
    std::string error;
};

namespace detail {

class JsonParser {
public:
    explicit JsonParser(std::string_view text) : m_t(text) {}

    Json::ParseResult run() {
        Json::ParseResult r;
        skipWs();
        if (!value(r.value, 0)) {
            r.value = Json();
            r.error = m_err;
            return r;
        }
        skipWs();
        if (m_i != m_t.size()) {
            fail("unexpected trailing characters");
            r.value = Json();
            r.error = m_err;
            return r;
        }
        r.ok = true;
        return r;
    }

private:
    static constexpr int kMaxDepth = 64;

    bool fail(const char* msg) {
        if (m_err.empty()) m_err = std::string(msg) + " at offset " + std::to_string(m_i);
        return false;
    }

    bool eof() const { return m_i >= m_t.size(); }

    void skipWs() {
        while (!eof()) {
            char c = m_t[m_i];
            if (c == ' ' || c == '\t' || c == '\n' || c == '\r') ++m_i;
            else break;
        }
    }

    bool literal(const char* word, Json v, Json& out) {
        std::string_view w(word);
        if (m_t.substr(m_i, w.size()) != w) return fail("invalid literal");
        m_i += w.size();
        out = std::move(v);
        return true;
    }

    bool value(Json& out, int depth) {
        if (depth > kMaxDepth) return fail("nesting too deep");
        if (eof()) return fail("unexpected end of input");
        switch (m_t[m_i]) {
            case '{': return object(out, depth);
            case '[': return array(out, depth);
            case '"': {
                std::string s;
                if (!string(s)) return false;
                out = Json(std::move(s));
                return true;
            }
            case 't': return literal("true", Json(true), out);
            case 'f': return literal("false", Json(false), out);
            case 'n': return literal("null", Json(nullptr), out);
            default: return number(out);
        }
    }

    bool number(Json& out) {
        size_t start = m_i;
        while (!eof()) {
            char c = m_t[m_i];
            if ((c >= '0' && c <= '9') || c == '-' || c == '+' || c == '.' || c == 'e' || c == 'E') ++m_i;
            else break;
        }
        if (m_i == start) return fail("unexpected character");
        std::string s(m_t.substr(start, m_i - start));
        char* end = nullptr;
        double d = std::strtod(s.c_str(), &end);
        if (end != s.c_str() + s.size()) {
            m_i = start;
            return fail("invalid number");
        }
        out = Json(d);
        return true;
    }

    static void appendUtf8(std::string& s, unsigned cp) {
        if (cp < 0x80) {
            s.push_back(static_cast<char>(cp));
        } else if (cp < 0x800) {
            s.push_back(static_cast<char>(0xC0 | (cp >> 6)));
            s.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
        } else if (cp < 0x10000) {
            s.push_back(static_cast<char>(0xE0 | (cp >> 12)));
            s.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
            s.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
        } else {
            s.push_back(static_cast<char>(0xF0 | (cp >> 18)));
            s.push_back(static_cast<char>(0x80 | ((cp >> 12) & 0x3F)));
            s.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
            s.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
        }
    }

    bool hex4(unsigned& out) {
        if (m_i + 4 > m_t.size()) return fail("truncated \\u escape");
        unsigned v = 0;
        for (int k = 0; k < 4; ++k) {
            char c = m_t[m_i + static_cast<size_t>(k)];
            v <<= 4;
            if (c >= '0' && c <= '9') v |= static_cast<unsigned>(c - '0');
            else if (c >= 'a' && c <= 'f') v |= static_cast<unsigned>(c - 'a' + 10);
            else if (c >= 'A' && c <= 'F') v |= static_cast<unsigned>(c - 'A' + 10);
            else return fail("invalid \\u escape");
        }
        m_i += 4;
        out = v;
        return true;
    }

    bool string(std::string& out) {
        ++m_i;  // opening quote
        while (true) {
            if (eof()) return fail("unterminated string");
            unsigned char c = static_cast<unsigned char>(m_t[m_i]);
            if (c == '"') {
                ++m_i;
                return true;
            }
            if (c < 0x20) return fail("control character in string");
            if (c != '\\') {
                out.push_back(static_cast<char>(c));
                ++m_i;
                continue;
            }
            ++m_i;
            if (eof()) return fail("unterminated escape");
            char e = m_t[m_i++];
            switch (e) {
                case '"': out.push_back('"'); break;
                case '\\': out.push_back('\\'); break;
                case '/': out.push_back('/'); break;
                case 'b': out.push_back('\b'); break;
                case 'f': out.push_back('\f'); break;
                case 'n': out.push_back('\n'); break;
                case 'r': out.push_back('\r'); break;
                case 't': out.push_back('\t'); break;
                case 'u': {
                    unsigned cp = 0;
                    if (!hex4(cp)) return false;
                    if (cp >= 0xD800 && cp <= 0xDBFF) {
                        if (m_t.substr(m_i, 2) != "\\u") return fail("lone surrogate");
                        m_i += 2;
                        unsigned lo = 0;
                        if (!hex4(lo)) return false;
                        if (lo < 0xDC00 || lo > 0xDFFF) return fail("invalid surrogate pair");
                        cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
                    } else if (cp >= 0xDC00 && cp <= 0xDFFF) {
                        return fail("lone surrogate");
                    }
                    appendUtf8(out, cp);
                    break;
                }
                default: --m_i; return fail("invalid escape");
            }
        }
    }

    bool array(Json& out, int depth) {
        ++m_i;  // [
        out = Json::array();
        skipWs();
        if (!eof() && m_t[m_i] == ']') {
            ++m_i;
            return true;
        }
        while (true) {
            skipWs();
            Json item;
            if (!value(item, depth + 1)) return false;
            out.push(std::move(item));
            skipWs();
            if (eof()) return fail("unterminated array");
            char c = m_t[m_i++];
            if (c == ']') return true;
            if (c != ',') {
                --m_i;
                return fail("expected ',' or ']'");
            }
        }
    }

    bool object(Json& out, int depth) {
        ++m_i;  // {
        out = Json::object();
        skipWs();
        if (!eof() && m_t[m_i] == '}') {
            ++m_i;
            return true;
        }
        while (true) {
            skipWs();
            if (eof() || m_t[m_i] != '"') return fail("expected string key");
            std::string key;
            if (!string(key)) return false;
            skipWs();
            if (eof() || m_t[m_i] != ':') return fail("expected ':'");
            ++m_i;
            skipWs();
            Json item;
            if (!value(item, depth + 1)) return false;
            out.set(std::move(key), std::move(item));
            skipWs();
            if (eof()) return fail("unterminated object");
            char c = m_t[m_i++];
            if (c == '}') return true;
            if (c != ',') {
                --m_i;
                return fail("expected ',' or '}'");
            }
        }
    }

    std::string_view m_t;
    size_t m_i = 0;
    std::string m_err;
};

}  // namespace detail

inline Json::ParseResult Json::parse(std::string_view text) {
    return detail::JsonParser(text).run();
}

}  // namespace nf
